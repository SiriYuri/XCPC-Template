template<u32 P>
DynModInt<P> findPrimitiveRoot() {
    DynModInt<P> i = 2;
    int k = __builtin_ctz(DynModInt<P>::mod() - 1);
    while (true) {
        if (power(i, (DynModInt<P>::mod() - 1) / 2) != 1) {
            break;
        }
        i += 1;
    }
    return power(i, (DynModInt<P>::mod() - 1) >> k);
}

DynModInt<P> primitiveRoot = findPrimitiveRoot<P>();

template<u32 P>
void dft(std::vector<DynModInt<P>> &a) {
    int n = a.size();
    
    if (int(rev.size()) != n) {
        int k = __builtin_ctz(n) - 1;
        rev.resize(n);
        for (int i = 0; i < n; i++) {
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << k;
        }
    }
    
    for (int i = 0; i < n; i++) {
        if (rev[i] < i) {
            std::swap(a[i], a[rev[i]]);
        }
    }

    if (roots<P>.size() < n) {
        int k = __builtin_ctz(roots<P>.size());
        roots<P>.resize(n);
        while ((1 << k) < n) {
            auto e = power(primitiveRoot<P>, 1 << (__builtin_ctz(DynModInt<P>::mod() - 1) - k - 1));
            for (int i = 1 << (k - 1); i < (1 << k); i++) {
                roots<P>[2 * i] = roots<P>[i];
                roots<P>[2 * i + 1] = roots<P>[i] * e;
            }
            k++;
        }
    }

    for (int k = 1; k < n; k *= 2) {
        for (int i = 0; i < n; i += 2 * k) {
            for (int j = 0; j < k; j++) {
                DynModInt<P> u = a[i + j];
                DynModInt<P> v = a[i + j + k] * roots<P>[k + j];
                a[i + j] = u + v;
                a[i + j + k] = u - v;
            }
        }
    }
}

template<u32 P>
void idft(std::vector<DynModInt<P>> &a) {
    int n = a.size();
    std::reverse(a.begin() + 1, a.end());
    dft(a);
    DynModInt<P> inv = (1 - (i64)DynModInt<P>::mod()) / n;
    for (int i = 0; i < n; i++) {
        a[i] *= inv;
    }
}

struct Poly : public std::vector<Z> {

    Poly(std::vector<Value> &&a) : std::vector<Value>(std::move(a)) {}

        std::vector<Value>::operator=(std::move(a));

    }
    
    Poly(const std::initializer_list<Value> &a) : std::vector<Value>(a) {}
    
    template<class InputIt, std::enable_if_t<!std::is_integral_v<InputIt>, int> = 0>
    Poly(InputIt first, InputIt last) : std::vector<Value>(first, last) {}
    
    template<class F, std::enable_if_t<std::is_invocable_v<F, int>, int> = 0>
    explicit Poly(int n, F f) : std::vector<Value>(n) {
        for (int i = 0; i < n; i++) {
            (*this)[i] = f(i);
        }
    }
    
    Poly shift(int k) const {
        if (k >= 0) {
            auto b = *this;
            b.insert(b.begin(), k, 0);
            return b;
        } else if (this->size() <= -k) {
            return Poly();
        } else {
            return Poly(this->begin() + (-k), this->end());
        }
    }

    Poly trunc(int k) const {
        Poly f = *this;
        f.resize(k);
        return f;
    }

    friend Poly operator + (const Poly &a, const Poly &b) {
        Poly res(std::max(a.size(), b.size()));
        for (int i = 0; i < a.size(); i++) res[i] += a[i];
        for (int i = 0; i < b.size(); i++) res[i] += b[i];
        return res;
    }

    friend Poly operator - (const Poly &a, const Poly &b) {
        Poly res(std::max(a.size(), b.size()));
        for (int i = 0; i < a.size(); i++) res[i] += a[i];
        for (int i = 0; i < b.size(); i++) res[i] -= b[i];
        return res;
    }

        std::vector<Value> res(a.size());

                }
            }
            return c;
        }

        b.resize(n);

    Poly deriv() const {
        if (this->empty()) return Poly();
        Poly res(this->size() - 1);
        for (int i = 0; i < this->size() - 1; ++i) {
            res[i] = (i + 1) * (*this)[i + 1];
        }
        return res;
    }

    Poly integr() const {
        Poly res(this->size() + 1);
        for (int i = 0; i < this->size(); ++i) {
            res[i + 1] = (*this)[i] / (i + 1);
        }
        return res;

        Poly x{this->empty() ? Value(0) : (*this)[0].inv()};
        int k = 1;
        while (k < m) {
            k *= 2;
            x = (x * (Poly{2} - trunc(k) * x)).trunc(k);
        }
        return x.trunc(m);
    }

    Poly log(int m) const {
        return (deriv() * inv(m)).integr().trunc(m);
    }

    Poly exp(int m) const {
        Poly x{1};
        int k = 1;
        while (k < m) {
            k *= 2;
            x = (x * (Poly{1} - x.log(k) + trunc(k))).trunc(k);
        }
        return x.trunc(m);
    }

    Poly pow(int k, int m) const {
        int i = 0;
        while (i < this->size() && (*this)[i] == Value(0)) i++;
        if (i == this->size() || 1LL * i * k >= m) return Poly(m);
        Value v = (*this)[i];
        auto f = shift(-i) * v.inv();
        return (f.log(m - i * k) * k).exp(m - i * k).shift(i * k) * ::power(v, k);
    }

    Poly sqrt(int m) const {
        Poly x{1};
        int k = 1;
        Value inv2 = Value(2).inv();
        while (k < m) {
            k *= 2;
            x = (x + (trunc(k) * x.inv(k)).trunc(k)) * inv2;
        }
        return x.trunc(m);
    }

    Poly mulT(Poly b) const {
        if (b.size() == 0) return Poly();
        int n = b.size();
        std::reverse(b.begin(), b.end());
        return ((*this) * b).shift(-(n - 1));
    }
    
    std::vector<Value> eval(std::vector<Value> x) const {
        if (this->size() == 0) return std::vector<Value>(x.size(), 0);
        const int n = std::max(x.size(), this->size());
        std::vector<Poly> q(4 * n);
        std::vector<Value> ans(x.size());
        x.resize(n);
        
        std::function<void(int, int, int)> build = [&](int p, int l, int r) {
            if (r - l == 1) {
                q[p] = Poly{1, -x[l]};
            } else {
                int m = (l + r) / 2;
                build(2 * p, l, m);
                build(2 * p + 1, m, r);
                q[p] = q[2 * p] * q[2 * p + 1];
            }
        };
        build(1, 0, n);
        
        std::function<void(int, int, int, const Poly &)> work = [&](int p, int l, int r, const Poly &num) {
            if (r - l == 1) {
                if (l < int(ans.size())) ans[l] = num[0];
            } else {
                int m = (l + r) / 2;
                work(2 * p, l, m, num.mulT(q[2 * p + 1]).trunc(m - l));
                work(2 * p + 1, m, r, num.mulT(q[2 * p]).trunc(r - m));
            }
        };
        work(1, 0, n, mulT(q[1].inv(n)));
        return ans;
    }
};
