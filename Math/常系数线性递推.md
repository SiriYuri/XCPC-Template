常系数线性递推  
初始项 a_0, a_1, ..., a_{k-1}
递推系数 f_1, f_2, ..., f_k (使得 a_i = f_1*a_{i-1} + ... + f_k*a_{i-k})
要求的项数 n (0-indexed)  
O(k log k log n), 瓶颈在于多项式操作, 朴素的是 O(k^2 logn)
```cpp
Z linearRecurrence(const std::vector<Z>& a, const std::vector<Z>& c, i64 n) {
    if (n < a.size()) return a[n];
    int k = c.size();
    if (k == 0) return 0;
    Poly P(k + 1);
    P[k] = 1;
    for (int i = 0; i < k; i++) {
        P[k - 1 - i] = -c[i];
    }
    Poly res{1}, base{0, 1}; 
    while (n > 0) {
        if (n & 1) res = (res * base) % P;
        base = (base * base) % P;
        n >>= 1;
    }
    Z ans = 0;
    for (int i = 0; i < std::min((int)res.size(), k); i++) {
        ans += res[i] * a[i];
    }
    return ans;
}
```
BM
给一个数列, 求其递推系数  
O(n^2)  
注意留出部分数据进行检验, 并且系数向量的大小越小正确性越高, 长度超过原数组一半可以近似认为错误
```cpp
std::vector<Z> BM(const std::vector<Z>& s) {
    std::vector<Z> C, B;
    int L = 0, m = 1;
    Z b = 1;
    for (int i = 0; i < s.size(); i++) {
        Z d = s[i];
        for (int j = 0; j < L; j++) {
            d -= C[j] * s[i - 1 - j];
        }
        if (d == 0) {
            m++; 
            continue;
        }
        auto T = C;
        Z coef = d / b;
        if (C.size() <= B.size() + m) {
            C.resize(B.size() + m, 0);
        }
        C[m - 1] += coef; 
        for (int j = 0; j < B.size(); j++) {
            C[j + m] -= coef * B[j];
        }
        if (2 * L <= i) {
            L = i + 1 - L;
            B = T; 
            b = d; 
            m = 1;
        } else {
            m++;
        }
    }
    C.resize(L);
    return C;
}
```
