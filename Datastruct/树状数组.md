template<typename T>
struct Fenwick {
    int n;
    std::vector<T> a;

    Fenwick (int _n) : n(_n), a(n + 1) {}

    void add(int x, T w) {
        for (; x <= n; x += x & -x) {
            a[x] += w;
        }
    }

    T sum(int x) {
        T ans {};
        for (; x >= 1; x -= x & -x) {
            ans += a[x];
        }
        return ans;
    }

    T rangesum(int l, int r) {
        return sum(r) - sum(l - 1);
    }
};
