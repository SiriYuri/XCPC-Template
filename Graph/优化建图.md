#include "bits/stdc++.h"

using u32 = unsigned int;
using u64 = unsigned long long;
using u128 = __uint128_t;

using i64 = long long;
using i128 = __int128_t;

using f32 = double;
using f64 = long double;

// namespace ranges = std::ranges;

constexpr i64 inf = std::numeric_limits<i64>::max() >> 2;

#define mid ((l + r) >> 1)
#define ls (p << 1)
#define rs (p << 1 | 1)

void solve() {
    int n, m, s;
    std::cin >> n >> m >> s;

    int shift = n << 2;
    std::vector<int> leaf(n + 1); // i 在树上的叶子编号
    std::vector<std::array<int, 2>> segt((n << 3) + 1);
    std::vector<std::vector<std::pair<i64, i64>>> edge((n << 3) + 1);
    auto build = [&] (auto &&self, int p, int l, int r) -> void {
        segt[p] = {l, r};
        if (l == r) {
            leaf[l] = p;
            return ;
        }
        edge[p].emplace_back(ls, 0);
        edge[p].emplace_back(rs, 0);
        edge[ls + shift].emplace_back(p + shift, 0);
        edge[rs + shift].emplace_back(p + shift, 0);
        self(self, ls, l, mid);
        self(self, rs, mid + 1, r);
        return ;
    };

    auto link = [&] (auto &&self, int p, int l, int r, int ql, int qr, int pos, i64 w, int op) -> void {
        if (ql > r || qr < l) return;          
        if (ql <= l && r <= qr) {
            if (op == 0) edge[pos].emplace_back(p, w);           
            else edge[p + shift].emplace_back(pos, w);            
            return;
        }
        self(self, ls, l, mid, ql, qr, pos, w, op);
        self(self, rs, mid + 1, r, ql, qr, pos, w, op);
    };

    build(build, 1, 1, n);
    while (m--) {
        int op;
        std::cin >> op;
        if (op == 1) {
            i64 u, v, w;
            std::cin >> u >> v >> w;
            edge[leaf[u]].emplace_back(leaf[v], w);
        } else {
            i64 p, l, r, w;
            std::cin >> p >> l >> r >> w;
            link(link, 1, 1, n, l, r, leaf[p], w, op - 2 == 0 ? 0 : 1);
        }
    }
    for (int i = 1; i <= n; i++) {
        int u = leaf[i], v = u + shift;
        edge[u].emplace_back(v, 0);
        edge[v].emplace_back(u, 0);
    }
    std::vector<i64> dis((n << 3) + 1, inf);
    dis[leaf[s]] = 0;
    std::priority_queue<std::pair<i64, i64>, std::vector<std::pair<i64, i64>>, std::greater<std::pair<i64, i64>>> pq;
    pq.emplace(0, leaf[s]);
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d != dis[u]) {
            continue;
        }
        for (auto [v, w] : edge[u]) {
            if (dis[v] > d + w) {
                dis[v] = d + w;
                pq.emplace(dis[v], v);
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        std::cout << (dis[leaf[i] + shift] == inf ? -1 : dis[leaf[i] + shift]) << " \n"[i == n];
    }
    return ;
}

int main() {
    std::cin.tie(nullptr) -> std::ios_base::sync_with_stdio(false);

    int t = 1;
    // std::cin >> t;

    while (t-->0) {
        solve();
    }

    return 0;
}
