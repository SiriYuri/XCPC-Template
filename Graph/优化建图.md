线段树优化建图  
最好 
`SegtGraph::Graph graph(n);`    
`const auto &edge = graph.edg`    
这样使用
```cpp
namespace SegtGraph {
    #define ls (p << 1)
    #define rs (p << 1 | 1)
    #define mid ((l + r) >> 1)

    struct Graph {
        int n, shift;
        std::vector<int> leaf;
        std::vector<std::vector<std::pair<int, i64>>> edge;

    private:
        void build(int p, int l, int r) {
            if (l == r) {
                leaf[l] = p;
                return ;
            }
            edge[p].emplace_back(ls, 0);
            edge[p].emplace_back(rs, 0);
            edge[ls + shift].emplace_back(p + shift, 0);
            edge[rs + shift].emplace_back(p + shift, 0);
            build(ls, l, mid);
            build(rs, mid + 1, r);
        }

        void link(int op, int p, int l, int r, int x, int y, int pos, i64 w = 1) {
            if (x > r || y < l) {
                return ;
            }
            if (x <= l && r <= y) {
                if (!op) { // 点 -> 区间
                    edge[pos].emplace_back(p, w);
                } else { // 区间 -> 点
                    edge[p + shift].emplace_back(pos, w);
                }
                return ;
            }
            link(op, ls, l, mid, x, y, pos, w);
            link(op, rs, mid + 1, r, x, y, pos, w);
        }
    
    public:
        Graph(int _n) : n(_n), shift(n << 2), leaf(n + 1), edge(n << 3 ^ 1) {
            build(1, 1, n);
            for (int i = 1; i <= n; i++) {
                int u = leaf[i], v = u + shift;
                edge[u].emplace_back(v, 0);
                edge[v].emplace_back(u, 0);
            }
        }

        void addPointToPoint(int u, int v, i64 w = 1) {
            edge[leaf[u]].emplace_back(leaf[v], w);
        }

        void addPointToSeg(int u, int l, int r, i64 w = 1) {
            link(0, 1, 1, n, l, r, leaf[u], w);
        }

        void addSegToPoint(int l, int r, int v, i64 w = 1) {
            link(1, 1, 1, n, l, r, leaf[v], w);
        }

        int get(int p) const {
            return leaf[p];
        }
    };
}
```
