以下两个是有向图   
欧拉路径: 一笔画完所有边   
要求存在一个点出度 = 入度 + 1 (起点), 一个点入度 = 出度 + 1(终点), 其余点出度 = 入度   
$O((n + m)log)$, 瓶颈在于排序, 不需要字典序最小则是 $O(n + m)$
```cpp
struct EulerPath {
    int n, m;
    std::vector<std::vector<int>> ver;
    std::vector<int> deg1, degO;
    std::vector<int> head; 
    std::vector<int> ans;  

    EulerPath(int n) : n(n), m(0) {
        ver.assign(n + 1, std::vector<int>());
        deg1.assign(n + 1, 0);
        degO.assign(n + 1, 0);
        head.assign(n + 1, 0);
    }

    void addEdge(int u, int v) {
        ver[u].push_back(v);
        deg1[v]++;
        degO[u]++;
        m++;
    }

    void dfs(int u) {
        while (head[u] < ver[u].size()) {
            int v = ver[u][head[u]++]; 
            dfs(v);
        }
        ans.push_back(u);
    }

    std::vector<int> work() {
        for (int i = 1; i <= n; i++) {
            std::sort(ver[i].begin(), ver[i].end());
        }
        int s = -1;
        int start = 0, end = 0;
        bool ok = true;
        int first = -1; 
        for (int i = 1; i <= n; i++) {
            if (deg1[i] > 0 || degO[i] > 0) {
                if (first == -1) first = i;
            }
            if (degO[i] - deg1[i] == 1) {
                s = i;
                start++;
            } else if (deg1[i] - degO[i] == 1) {
                end++;
            } else if (deg1[i] != degO[i]) {
                ok = false;
            }
        }
        if (!ok || !((start == 0 && end == 0) || (start == 1 && end == 1))) {
            return {}; 
        }
        if (s == -1) {
            s = (first != -1 ? first : 1);
        }
        dfs(s);
        if (ans.size() != m + 1) {
            return {}; 
        }
        std::reverse(ans.begin(), ans.end());
        return ans;
    }
};
```

```cpp
            }
        }
        if (s == -1) {
            s = 1; 
        }
        dfs(s);
        if (ans.size() != m + 1) {
            return {}; 
        }
        std::reverse(ans.begin(), ans.end());
        return ans;
    }
```

以下两个是无向图   
欧拉路径: 要求恰好两个点度数为奇数 (起点, 终点), 其余点度数为偶数
```cpp
struct EulerPath {
    struct Edge {
        int to;
        int id; 
        
        bool operator < (const Edge& other) const {
            return to < other.to;
        }
    };

    int n, m;
    std::vector<std::vector<Edge>> ver;
    std::vector<int> deg;      
    std::vector<int> head;     
    std::vector<bool> vis;
    std::vector<int> ans;

    EulerPath(int n) : n(n), m(0) {
        ver.assign(n + 1, std::vector<Edge>());
        deg.assign(n + 1, 0);
        head.assign(n + 1, 0);
    }

    void addEdge(int u, int v) {
        ver[u].push_back({v, m});
        ver[v].push_back({u, m});
        deg[u]++;
        deg[v]++;
        vis.push_back(false); 
        m++;
    }

    void dfs(int u) {
        while (head[u] < ver[u].size()) {
            auto edge = ver[u][head[u]++]; 
            if (vis[edge.id]) {
                continue;
            }
            vis[edge.id] = true; 
            dfs(edge.to);
        }
        ans.push_back(u); 
    }

    std::vector<int> work() {
        for (int i = 1; i <= n; i++) {
            std::sort(ver[i].begin(), ver[i].end());
        }
        int s = -1;
        int odd = 0;
        int first = -1;
        for (int i = 1; i <= n; i++) {
            if (deg[i] > 0 && first == -1) {
                first = i;
            }
            if (deg[i] % 2 != 0) {
                odd++;
                if (s == -1) {
                    s = i;
                }
            }
        }
        if (odd != 0 && odd != 2) {
            return {};
        }
        if (s == -1) {
            s = (first != -1 ? first : 1);
        }
        dfs(s);
        if (ans.size() != m + 1) {
            return {};
        } 
        std::reverse(ans.begin(), ans.end());
        return ans;
    }
};
```
