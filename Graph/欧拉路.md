欧拉路径: 一笔画完所有边   
要求存在一个点出度 = 入度 + 1 (起点), 一个点入度 = 出度 + 1(终点), 其余点出度 = 入度   
$O((n + m)log)$, 瓶颈在于排序, 不需要字典序最小则是 $O(n + m)$
```cpp
struct EulerPath {
    int n, m;
    std::vector<std::vector<int>> ver;
    std::vector<int> deg1, degO;
    std::vector<int> head; 
    std::vector<int> ans;  

    EulerPath(int n) : n(n), m(0) {
        ver.assign(n + 1, std::vector<int>());
        deg1.assign(n + 1, 0);
        degO.assign(n + 1, 0);
        head.assign(n + 1, 0);
    }

    void addEdge(int u, int v) {
        ver[u].push_back(v);
        deg1[v]++;
        degO[u]++;
        m++;
    }

    void dfs(int u) {
        while (head[u] < ver[u].size()) {
            int v = ver[u][head[u]++]; 
            dfs(v);
        }
        ans.push_back(u);
    }

    std::vector<int> work() {
        for (int i = 1; i <= n; i++) {
            std::sort(ver[i].begin(), ver[i].end());
        }
        int s = -1;
        int start = 0, end = 0;
        bool ok = true;
        int first = -1; 
        for (int i = 1; i <= n; i++) {
            if (deg1[i] > 0 || degO[i] > 0) {
                if (first == -1) first = i;
            }
            if (degO[i] - deg1[i] == 1) {
                s = i;
                start++;
            } else if (deg1[i] - degO[i] == 1) {
                end++;
            } else if (deg1[i] != degO[i]) {
                ok = false;
            }
        }
        if (!ok || !((start == 0 && end == 0) || (start == 1 && end == 1))) {
            return {}; 
        }
        if (s == -1) {
            s = (first != -1 ? first : 1);
        }
        dfs(s);
        if (ans.size() != m + 1) {
            return {}; 
        }
        std::reverse(ans.begin(), ans.end());
        return ans;
    }
};
```

需要所有点满足 入度 = 出度

struct DirectedEulerCircuit {
    int n, m;
    std::vector<std::vector<int>> ver;
    std::vector<int> deg1, degO; 
    std::vector<int> head; 
    std::vector<int> ans;  

    DirectedEulerCircuit(int n) : n(n), m(0) {
        ver.assign(n + 1, std::vector<int>());
        deg1.assign(n + 1, 0);
        degO.assign(n + 1, 0);
        head.assign(n + 1, 0);
    }

    void addEdge(int u, int v) {
        ver[u].push_back(v);
        deg1[v]++;
        degO[u]++;
        m++;
    }

    void dfs(int u) {
        while (head[u] < ver[u].size()) {
            int v = ver[u][head[u]++]; 
            dfs(v);
        }
        ans.push_back(u);
    }

    std::vector<int> work() {
        for (int i = 1; i <= n; i++) {
            std::sort(ver[i].begin(), ver[i].end());
        }
        int s = -1;
        for (int i = 1; i <= n; i++) {
            if (deg1[i] != degO[i]) {
                return {}; 
            }
            if (s == -1 && degO[i] > 0) {
                s = i;
            }
        }
        if (s == -1) {
            s = 1; 
        }
        dfs(s);
        if (ans.size() != m + 1) {
            return {}; 
        }
        std::reverse(ans.begin(), ans.end());
        return ans;
    }
};
