将原图的点双视作方点, 初始点视作原点重构的图   
图一定是圆方交错的, 用来快速查询路径信息   
```cpp
struct BlockCutTree {
    int n, cnt, cur;
    std::vector<std::vector<int>> adj;
    std::vector<int> dfn, low, stk;
    std::vector<std::pair<int, int>> edges;
    
    BlockCutTree(int _n) : n(_n), adj(n + 1), dfn(n + 1, -1), low(n + 1), cnt(0), cur(1) {}
    
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    void dfs(int x, int parent) {
        stk.push_back(x);
        dfn[x] = low[x] = cur++;
        for (auto y : adj[x]) {
            if (y == parent) {
                continue;
            } 
            if (dfn[y] == -1) {
                dfs(y, x);
                low[x] = std::min(low[x], low[y]);
                if (low[y] >= dfn[x]) {
                    cnt++; 
                    int v;
                    do {
                        v = stk.back();
                        stk.pop_back();
                        edges.emplace_back(n + cnt, v); 
                    } while (v != y);
                    edges.emplace_back(x, n + cnt); 
                }
            } else {
                low[x] = std::min(low[x], dfn[y]);
            }
        }
    }
    
    std::pair<int, std::vector<std::pair<int, int>>> work() {
        for (int i = 1; i <= n; i++) {
            if (dfn[i] == -1) {
                stk.clear();
                dfs(i, 0);
            }
        }
        return {cnt, edges};
    }
};
```
