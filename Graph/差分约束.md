给出一组包含 m 个不等式，有 n 个未知数的形如:
$$
x_{c1}​ ​− x_{c1′} ​​≤ y_1
​xc2​​−xc2′​​≤y2
​⋯
xcm​​−xcm′​​≤ym​​
$$
的不等式组，求任意一组满足这个不等式组的解
```cpp
void solve() {
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<std::pair<int, int>>> edge(n + 1);
    for (int i = 1; i <= n; i++) {
        edge[0].emplace_back(i, 0);
    }
    for (int i = 1; i <= m; i++) {
        int u, v, w;
        std::cin >> u >> v >> w;
        edge[v].emplace_back(u, w);
    }
    std::queue<int> que;
    std::vector<int> dis(n + 1, inf<int>);
    std::vector<int> cnt(n + 1), vis(n + 1);  
    dis[0] = 0;
    vis[0] = 1;
    que.emplace(0);
    while(!que.empty()) {
        int u = que.front();
        que.pop();
        vis[u] = 0;
        for (auto &[v, w] : edge[u]) {
            if (dis[v] > dis[u] + w) {
                dis[v] = dis[u] + w;
                cnt[v] = cnt[u] + 1;
                if (cnt[v] >= n + 1) {
                    std::cout << "NO\n";
                    return ;
                }
                if (!vis[v]) {
                    vis[v] = 1;
                    que.emplace(v);
                }
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        std::cout << dis[i] << " \n"[i == n];
    }
    return ;
}
```
