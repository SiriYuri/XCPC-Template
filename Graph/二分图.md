最大匹配   
$O($\sqrtn$m)$

```cpp
struct HopcroftKarp {
    int n, m; 
    std::vector<std::array<int, 2>> edges;
    std::vector<int> matchl, matchr;

    HopcroftKarp(int _n, int _m) : n(_n), m(_m) {
        matchl.assign(n, -1);
        matchr.assign(m, -1);
    }

    void addEdge(int u, int v) {
        edges.push_back({u - 1, v - 1});
    }

    int work() {
        std::vector<int> adj(edges.size());
        std::mt19937 rng(std::chrono::steady_clock::now().time_since_epoch().count());
        std::shuffle(edges.begin(), edges.end(), rng);
        std::vector<int> head(n + 1, 0);
        for (auto& [u, v] : edges) {
            head[u]++;
        }
        for (int i = 1; i <= n; i++) {
            head[i] += head[i - 1];
        }
        for (auto& [u, v] : edges) {
            adj[--head[u]] = v;
        }
        int match = 0;
        std::vector<int> root, parent, que(n);
        while (true) {
            root.assign(n, -1);
            parent.assign(n, -1);
            int tail = 0;
            for (int i = 0; i < n; i++) {
                if (matchl[i] == -1) {
                    que[tail++] = i;
                    root[i] = i;
                    parent[i] = i;
                }
            }
            bool find = false;
            for (int i = 0; i < tail; i++) {
                int curr_u = que[i];
                if (matchl[root[curr_u]] != -1) {
                    continue;
                }
                for (int j = head[curr_u]; j < head[curr_u + 1]; j++) {
                    int cur = adj[j];
                    if (matchr[cur] == -1) {
                        while (cur != -1) {
                            matchr[cur] = curr_u;
                            std::swap(matchl[curr_u], cur);
                            curr_u = parent[curr_u];
                        }
                        find = true;
                        match++;
                        break;
                    }
                    if (parent[matchr[cur]] == -1) {
                        int next_u = matchr[cur];
                        que[tail++] = next_u;
                        parent[next_u] = curr_u;
                        root[next_u] = root[curr_u];
                    }
                }
            }
            if (!find) {
                break;
            }
        }
        return match;
    }

    std::vector<std::array<int, 2>> get() {
        std::vector<std::array<int, 2>> ans;
        for (int i = 0; i < n; i++) {
            if (matchl[i] != -1) {
                ans.push_back({i + 1, matchl[i] + 1});
            }
        }
        return ans;
    }
};
```

带权的   
$O(n^3)$
```cpp
template<typename T>
struct KM {
    int n;
    std::vector<int> ansl, ansr, pre;
    std::vector<T> lx, ly;
    std::vector<std::vector<T>> ver;
    
    const T inf = std::numeric_limits<T>::max() >> 2; 

    KM(int n) : n(n) {
        ver.assign(n + 1, std::vector<T>(n + 1, -inf));
        ansl.assign(n + 1, -1);
        ansr.assign(n + 1, -1);
        lx.assign(n + 1, -inf);
        ly.assign(n + 1, 0);
        pre.assign(n + 1, -1);
    }

    void addEdge(int x, int y, T w) {
        ver[x][y] = std::max(ver[x][y], w); 
    }

    void bfs(int st) {
        std::vector<bool> visl(n + 1, false), visr(n + 1, false);
        std::vector<T> slack(n + 1, inf);
        std::queue<int> que;

        auto check = [&](int x) -> bool {
            visr[x] = true;
            if (ansr[x] != -1) {
                que.push(ansr[x]);
                visl[ansr[x]] = true;
                return false;
            }
            while (x != -1) {
                ansr[x] = pre[x];
                std::swap(x, ansl[pre[x]]);
            }
            return true;
        };

        que.push(st);
        visl[st] = true;
        while (true) {
            while (!que.empty()) {
                int x = que.front();
                que.pop();
                for (int y = 1; y <= n; ++y) {
                    if (visr[y]) continue;
                    T del = lx[x] + ly[y] - ver[x][y];
                    if (del < slack[y]) {
                        pre[y] = x;
                        slack[y] = del;
                        if (slack[y] == 0 && check(y)) return;
                    }
                }
            }
            T val = inf;
            for (int i = 1; i <= n; ++i) {
                if (!visr[i]) {
                    val = std::min(val, slack[i]);
                }
            }
            for (int i = 1; i <= n; ++i) {
                if (visl[i]) lx[i] -= val;
                if (visr[i]) {
                    ly[i] += val;
                } else {
                    slack[i] -= val;
                }
            }
            for (int i = 1; i <= n; ++i) {
                if (!visr[i] && slack[i] == 0 && check(i)) {
                    return;
                }
            }
        }
    }

    T work() {
        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= n; ++j) {
                lx[i] = std::max(lx[i], ver[i][j]);
            }
        }
        for (int i = 1; i <= n; ++i) {
            bfs(i);
        }
        T res = 0;
        for (int i = 1; i <= n; ++i) {
            res += ver[ansr[i]][i]; 
        }
        return res;
    }

    std::vector<std::pair<int, int>> get() const {
        std::vector<std::pair<int, int>> edges;
        for (int i = 1; i <= n; ++i) {
            if (ansr[i] != -1 && ver[ansr[i]][i] != -inf) {
                edges.emplace_back(ansr[i], i);
            }
        }
        return edges;
    }
};
```
