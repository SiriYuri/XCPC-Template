求一般图的最大匹配
最大匹配指的是从边集中选出一个没有共用顶点的子集
$O(n^3)$
```cpp
struct Graph {
    int n, tim;
    std::vector<std::vector<int>> edge;
    std::vector<int> match, pre, f, type, st;
    std::queue<int> que;

    Graph(int _n) : n(_n), edge(n + 1), match(n + 1), pre(n + 1), f(n + 1), type(n + 1), st(n + 1), tim(0) {}

    void addEdge(int u, int v) {
        edge[u].emplace_back(v);
        edge[v].emplace_back(u);
    }

    int find(int x) {
        return f[x] == x ? x : f[x] = find(f[x]);
    }

    int lca(int u, int v) {
        ++tim;
        u = find(u);
        v = find(v);
        while (u || v) {
            if (u) {
                if (st[u] == tim) {
                    return u;
                }
                st[u] = tim;
                u = find(pre[match[u]]);
            }
            std::swap(u, v);
        }
        return -1;
    }

    void shrink(int u, int v, int l) {
        while (find(u) != l) {
            pre[u] = v;
            v = match[u];
            if (type[v] == 2) {
                type[v] = 1;
                que.emplace(v);
            }
            if (f[u] == u) {
                f[u] = l;
            }
            if (f[v] == v) {
                f[v] = l;
            }
            u = pre[v];
        }
    }

    bool bfs(int s) {
        std::iota(f.begin(), f.end(), 0);
        std::fill(type.begin(), type.end(), 0);
        std::fill(pre.begin(), pre.end(), 0);
        while (!que.empty()) {
            que.pop();
        }
        que.emplace(s);
        type[s] = 1;
        while (!que.empty()) {
            int u = que.front();
            que.pop();
            for (auto v : edge[u]) {
                if (find(u) == find(v)) {
                    continue;
                }
                if (!type[v]) {
                    type[v] = 2;
                    pre[v] = u;
                    if (!match[v]) {
                        int cur = v;
                        while (cur) {
                            int last = pre[cur];
                            int nxt = match[last];
                            match[cur] = last;
                            match[last] = cur;
                            cur = nxt;
                        }
                        return 1;
                    } else {
                        type[match[v]] = 1;
                        que.emplace(match[v]);
                    }
                } 
                else if (type[v] == 1) {
                    int l = lca(u, v);
                    shrink(u, v, l);
                    shrink(v, u, l);
                }
            } 
        }
        return 0;
    }

    std::vector<int> work() {
        for (int i = 1; i <= n; i++) {
            if (!match[i]) {
                for (auto v : edge[i]) {
                    if (!match[v]) {
                        match[i] = v;
                        match[v] = i;
                        break;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (!match[i]) {
                bfs(i);
            }
        }
        return match;
    }
};
```
