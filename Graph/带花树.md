求一般图的最大匹配   
最大匹配指的是从边集中选出一个没有共用顶点的子集   
$O(n^3)$
```cpp
struct BloosmTree {
    int n, tim;
    std::vector<std::vector<int>> edge;
    std::vector<int> match, pre, f, type, st;
    std::queue<int> que;

    BloosmTree(int _n) : n(_n), edge(n + 1), match(n + 1), pre(n + 1), f(n + 1), type(n + 1), st(n + 1), tim(0) {}

    void addEdge(int u, int v) {
        edge[u].emplace_back(v);
        edge[v].emplace_back(u);
    }

    int find(int x) {
        return f[x] == x ? x : f[x] = find(f[x]);
    }

    int lca(int u, int v) {
        ++tim;
        u = find(u);
        v = find(v);
        while (u || v) {
            if (u) {
                if (st[u] == tim) {
                    return u;
                }
                st[u] = tim;
                u = find(pre[match[u]]);
            }
            std::swap(u, v);
        }
        return -1;
    }

    void shrink(int u, int v, int l) {
        while (find(u) != l) {
            pre[u] = v;
            v = match[u];
            if (type[v] == 2) {
                type[v] = 1;
                que.emplace(v);
            }
            if (f[u] == u) {
                f[u] = l;
            }
            if (f[v] == v) {
                f[v] = l;
            }
            u = pre[v];
        }
    }

    bool bfs(int s) {
        std::iota(f.begin(), f.end(), 0);
        std::fill(type.begin(), type.end(), 0);
        std::fill(pre.begin(), pre.end(), 0);
        while (!que.empty()) {
            que.pop();
        }
        que.emplace(s);
        type[s] = 1;
        while (!que.empty()) {
            int u = que.front();
            que.pop();
            for (auto v : edge[u]) {
                if (find(u) == find(v)) {
                    continue;
                }
                if (!type[v]) {
                    type[v] = 2;
                    pre[v] = u;
                    if (!match[v]) {
                        int cur = v;
                        while (cur) {
                            int last = pre[cur];
                            int nxt = match[last];
                            match[cur] = last;
                            match[last] = cur;
                            cur = nxt;
                        }
                        return 1;
                    } else {
                        type[match[v]] = 1;
                        que.emplace(match[v]);
                    }
                } 
                else if (type[v] == 1) {
                    int l = lca(u, v);
                    shrink(u, v, l);
                    shrink(v, u, l);
                }
            } 
        }
        return 0;
    }

    std::vector<int> work() {
        for (int i = 1; i <= n; i++) {
            if (!match[i]) {
                for (auto v : edge[i]) {
                    if (!match[v]) {
                        match[i] = v;
                        match[v] = i;
                        break;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (!match[i]) {
                bfs(i);
            }
        }
        return match;
    }
};
```

```cpp
template <typename T = i64>
struct BlossomTree {
    struct Edge {
        int u, v;
        T w;
    };

    int n, m, time_stamp;
    const T inf = std::numeric_limits<T>::max() / 2;

    std::vector<std::vector<Edge>> adj;
    std::vector<std::vector<int>> base, blossom_nodes;
    std::vector<T> label;
    std::vector<int> match, slack, top, parent, color, visited_time;
    std::queue<int> que;

    BlossomTree(int _n) : n(_n) {
        int N = 2 * n + 2; 
        adj.assign(N, std::vector<Edge>(N, {0, 0, 0}));
        base.assign(N, std::vector<int>(N, 0));
        blossom_nodes.assign(N, std::vector<int>());
        label.assign(N, 0);
        match.assign(N, 0); slack.assign(N, 0); top.assign(N, 0);
        parent.assign(N, 0); color.assign(N, 0); visited_time.assign(N, 0);
    }

    T calc_delta(const Edge& cur) {
        return label[cur.u] + label[cur.v] - cur.w * 2;
    }

    void update_slack(int u, int v) {
        if (!slack[v] || calc_delta(adj[u][v]) < calc_delta(adj[slack[v]][v])) {
            slack[v] = u;
        }
    }

    void set_slack(int v) {
        slack[v] = 0;
        for (int u = 1; u <= n; ++u) {
            if (adj[u][v].w > 0 && top[u] != v && !color[top[u]]) {
                update_slack(u, v);
            }
        }
    }

    void enqueue(int u) {
        if (u <= n) {
            que.emplace(u);
        } else {
            for (int v : blossom_nodes[u]) enqueue(v);
        }
    }

    void set_top(int u, int w) {
        top[u] = w;
        if (u > n) {
            for (int v : blossom_nodes[u]) set_top(v, w);
        }
    }

    int get_pos(int u, int v) {
        int pos = std::find(blossom_nodes[u].begin(), blossom_nodes[u].end(), v) - blossom_nodes[u].begin();
        if (pos % 2 == 1) {
            std::reverse(blossom_nodes[u].begin() + 1, blossom_nodes[u].end());
            return (int)blossom_nodes[u].size() - pos;
        }
        return pos;
    }

    void set_match(int u, int v) {
        match[u] = adj[u][v].v;
        if (u <= n) return;
        Edge w = adj[u][v];
        int curr_x = base[u][w.u];
        int curr_y = get_pos(u, curr_x);
        for (int i = 0; i < curr_y; i++) {
            set_match(blossom_nodes[u][i], blossom_nodes[u][i ^ 1]);
        }
        set_match(curr_x, v);
        std::rotate(blossom_nodes[u].begin(), blossom_nodes[u].begin() + curr_y, blossom_nodes[u].end());
    }

    void augment(int u, int v) {
        int w = top[match[u]];
        set_match(u, v);
        if (!w) return;
        set_match(w, top[parent[w]]);
        augment(top[parent[w]], w);
    }

    int lca(int u, int v) {
        for (++time_stamp; u || v; std::swap(u, v)) {
            if (!u) continue;
            if (visited_time[u] == time_stamp) return u;
            visited_time[u] = time_stamp;
            if ((u = top[match[u]])) u = top[parent[u]];
        }
        return 0;
    }

    void add_blossom(int u, int lca_node, int v) {
        int blossom = n + 1;
        while (blossom <= m && top[blossom]) ++blossom;
        if (blossom > m) ++m;
        label[blossom] = color[blossom] = top[blossom] = 0;
        match[blossom] = match[lca_node];
        blossom_nodes[blossom].clear();
        blossom_nodes[blossom].emplace_back(lca_node);
        
        int j;
        for (int i = u; i != lca_node; i = top[parent[j]]) {
            blossom_nodes[blossom].emplace_back(i);
            blossom_nodes[blossom].emplace_back(j = top[match[i]]);
            enqueue(j);
        }
        std::reverse(blossom_nodes[blossom].begin() + 1, blossom_nodes[blossom].end());
        for (int i = v; i != lca_node; i = top[parent[j]]) {
            blossom_nodes[blossom].emplace_back(i);
            blossom_nodes[blossom].emplace_back(j = top[match[i]]);
            enqueue(j);
        }
        set_top(blossom, blossom);
        for (int i = 1; i <= m; ++i) {
            adj[blossom][i].w = adj[i][blossom].w = 0;
        }
        std::fill(base[blossom].begin() + 1, base[blossom].begin() + n + 1, 0);
        for (int curr_u : blossom_nodes[blossom]) {
            for (int curr_v = 1; curr_v <= m; ++curr_v) {
                if (!adj[blossom][curr_v].w || calc_delta(adj[curr_u][curr_v]) < calc_delta(adj[blossom][curr_v])) {
                    adj[blossom][curr_v] = adj[curr_u][curr_v];
                    adj[curr_v][blossom] = adj[curr_v][curr_u];
                }
            }
            for (int curr_v = 1; curr_v <= n; ++curr_v) {
                if (base[curr_u][curr_v]) base[blossom][curr_v] = curr_u;
            }
        }
        set_slack(blossom);
    }

    void expand_blossom(int blossom) {
        for (int curr : blossom_nodes[blossom]) set_top(curr, curr);
        int lca_node = base[blossom][adj[blossom][parent[blossom]].u];
        int rot_pos = get_pos(blossom, lca_node);
        for (int i = 0; i < rot_pos; i += 2) {
            int curr_x = blossom_nodes[blossom][i], curr_y = blossom_nodes[blossom][i + 1];
            parent[curr_x] = adj[curr_y][curr_x].u;
            color[curr_x] = 1;
            color[curr_y] = slack[curr_x] = 0;
            set_slack(curr_y);
            enqueue(curr_y);
        }
        color[lca_node] = 1;
        parent[lca_node] = parent[blossom];
        for (int i = rot_pos + 1; i < (int)blossom_nodes[blossom].size(); i++) {
            color[blossom_nodes[blossom][i]] = -1;
            set_slack(blossom_nodes[blossom][i]);
        }
        top[blossom] = 0;
    }

    bool process_edge(const Edge& cur) {
        int u = top[cur.u], v = top[cur.v];
        if (color[v] == -1) {
            parent[v] = cur.u;
            color[v] = 1;
            int matched_v = top[match[v]];
            slack[v] = slack[matched_v] = color[matched_v] = 0;
            enqueue(matched_v);
        } else if (!color[v]) {
            int lca_node = lca(u, v);
            if (!lca_node) {
                augment(u, v); 
                augment(v, u); 
                return true;
            } else {
                add_blossom(u, lca_node, v);
            }
        }
        return false;
    }

    bool bfs() {
        std::fill(color.begin() + 1, color.begin() + m + 1, -1);
        std::fill(slack.begin() + 1, slack.begin() + m + 1, 0);
        while (!que.empty()) que.pop();
        for (int i = 1; i <= m; ++i) {
            if (top[i] == i && !match[i]) {
                parent[i] = color[i] = 0;
                enqueue(i);
            }
        }
        if (que.empty()) return false;
        while (true) {
            while (!que.empty()) {
                int u = que.front(); 
                que.pop();
                if (color[top[u]] == 1) continue;
                for (int v = 1; v <= n; ++v) {
                    if (adj[u][v].w > 0 && top[u] != top[v]) {
                        if (calc_delta(adj[u][v]) != 0) {
                            update_slack(u, top[v]);
                        } else if (process_edge(adj[u][v])) {
                            return true;
                        }
                    }
                }
            }
            T delta = inf;
            for (int i = n + 1; i <= m; ++i) {
                if (top[i] == i && color[i] == 1) {
                    delta = std::min(delta, label[i] / 2);
                }
            }
            for (int i = 1; i <= m; ++i) {
                if (top[i] == i && slack[i] && color[i] != 1) {
                    T current_delta = calc_delta(adj[slack[i]][i]);
                    if (color[i] == -1) {
                        delta = std::min(delta, current_delta);
                    } else if (color[i] == 0) {
                        delta = std::min(delta, current_delta / 2);
                    }
                }
            }
            for (int i = 1; i <= n; ++i) {
                if (~color[top[i]]) {
                    if ((label[i] += (color[top[i]] * 2 - 1) * delta) <= 0) return false;
                }
            }
            for (int i = n + 1; i <= m; ++i) {
                if (top[i] == i && ~color[top[i]]) {
                    label[i] += (2 - color[top[i]] * 4) * delta;
                }
            }
            while (!que.empty()) que.pop();
            for (int i = 1; i <= m; ++i) {
                if (top[i] == i && slack[i] && top[slack[i]] != i && calc_delta(adj[slack[i]][i]) == 0 && process_edge(adj[slack[i]][i])) {
                    return true;
                }
            }
            for (int i = n + 1; i <= m; ++i) {
                if (top[i] == i && color[i] == 1 && !label[i]) expand_blossom(i);
            }
        }
        return false;
    }

    std::pair<T, std::vector<int>> work(const std::vector<std::tuple<int, int, T>>& edges) {
        m = n;
        time_stamp = 0;
        std::fill(visited_time.begin(), visited_time.end(), 0);
        std::fill(match.begin(), match.end(), 0);
        std::iota(top.begin() + 1, top.begin() + n + 1, 1);
        
        T max_edge_weight = 0;
        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= n; ++j) {
                adj[i][j] = {i, j, 0};
            }
        }
        for (auto [u, v, w] : edges) {
            max_edge_weight = std::max(max_edge_weight, w);
            adj[u][v].w = std::max(adj[u][v].w, w);
            adj[v][u].w = std::max(adj[v][u].w, w);
        }
        for (int i = 1; i <= n; ++i) {
            blossom_nodes[i].clear();
        }
        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= n; ++j) {
                base[i][j] = (i == j ? i : 0);
            }
        }
        
        std::fill(label.begin() + 1, label.begin() + n + 1, max_edge_weight);
        while (bfs()) {}
        
        T max_weight_sum = 0;
        std::vector<int> match_res(n + 1, 0);
        for (int i = 1; i <= n; ++i) {
            if (match[i]) {
                max_weight_sum += adj[i][match[i]].w;
                match_res[i] = match[i];
            }
        }
        return {max_weight_sum / 2, match_res};
    }
};
```
