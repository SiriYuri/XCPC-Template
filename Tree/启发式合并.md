```cpp
#include "bits/stdc++.h"

using i64 = long long;
using i128 = __int128_t;

using u32 = unsigned int;
using u64 = unsigned long long;
using u128 = __uint128_t;

namespace ranges = std::ranges;

void solve() {
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> edge(n + 1);
    for (int i = 1; i < n; i++) {
        int u, v;
        std::cin >> u >> v;
        edge[u].emplace_back(v);
        edge[v].emplace_back(u);
    }
    std::vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        std::cin >> a[i];
    }
    std::vector<int> siz(n + 1), son(n + 1);
    auto dfs1 = [&] (this auto &&self, int p, int fa) -> void {
        siz[p] = 1;
        for (auto c : edge[p]) {
            if (c != fa) {
                self(c, p);
                siz[p] += siz[c];
                if (siz[c] > siz[son[p]]) {
                    son[p] = c;
                }
            }
        }
    };

    dfs1(1, 0);
    int res = 0;
    std::vector<int> cnt(n + 1), ans(n + 1);
    auto add = [&] (int x) -> void {
        res += !cnt[x];
        cnt[x]++;
    };

    auto del = [&] (int x) -> void {
        res -= cnt[x] == 1;
        cnt[x]--;
    };

    auto work = [&] (this auto &&self, int p, int fa, bool keep) -> void {
        if (keep) {
            add(a[p]);
        } else {
            del(a[p]);
        }
        for (auto c : edge[p]) {
            if (c != fa) {
                self(c, p, keep);
            }
        }
    };

    auto dfs2 = [&] (this auto &&self, int p, int fa, bool keep) -> void {
        for (auto c : edge[p]) {
            if (c != fa && c != son[p]) {
                self(c, p, 0); 
            }
        }
        if (son[p]) {
            self(son[p], p, 1); 
        }
        add(a[p]); 
        for (auto c : edge[p]) {
            if (c != fa && c != son[p]) {
                work(c, p, 1); 
            }
        }
        ans[p] = res; 
        if (!keep) {
            work(p, fa, 0); 
        }
    };

    dfs2(1, 1, 0);
    int q;
    std::cin >> q;
    while (q--) {
        int x;
        std::cin >> x;
        std::cout << ans[x] << "\n";
    }
    return ;
}

int main() {
    std::cin.tie(nullptr) -> std::ios_base::sync_with_stdio(false);

    int t = 1;
    // std::cin >> t;

    while (t-->0) {
        solve();
    }

    return 0;
}
```
