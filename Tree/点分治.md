按重心分治, 每次计算经过根的贡献   
这是多次求是否存在长为 k 的路径
```cpp
#include "bits/stdc++.h"

using i64 = long long;
using i128 = __int128_t;

using u32 = unsigned int;
using u64 = unsigned long long;
using u128 = __uint128_t;

namespace ranges = std::ranges;

void solve() {
    int n, q;
    std::cin >> n >> q;
    std::vector<std::vector<std::pair<int, int>>> edge(n + 1);
    for (int i = 1; i < n; i++) {
        int u, v, w;
        std::cin >> u >> v >> w;
        edge[u].emplace_back(v, w);
        edge[v].emplace_back(u, w);
    }
    std::vector<int> qry(q + 1);
    for (int i = 1; i <= q; i++) {
        std::cin >> qry[i];
    }

    std::vector<int> has(10'000'001), vis(n + 1), siz(n + 1);
    auto findCd = [&] (int root) -> int {
        int tot = 0;
        auto dfs1 = [&] (this auto &&self, int p, int fa) -> void { // 获取 size
            siz[p] = 1;
            tot++;
            for (auto [c, w] : edge[p]) {
                if (c != fa && !vis[c]) {
                    self(c, p);
                    siz[p] += siz[c];
                }
            }
        };
        
        dfs1(root, 0);
        int rt = 0, minmax = n + 1;
        auto dfs2 = [&] (this auto &&self, int p, int fa) -> void { // 获取重心
            int mx = tot - siz[p];
            for (auto [c, w] : edge[p]) {
                if (c != fa && !vis[c]) {
                    self(c, p);
                    mx = std::max(mx, siz[c]);
                }
            }
            if (mx < minmax) {
                minmax = mx;
                rt = p;
            }
        };

        dfs2(root, 0);
        return rt;
    };
    
    std::vector<int> dis;
    auto update = [&] (this auto &&self, int p, int fa, int d) -> void {
        if (d > 10'000'000) {
            return ;
        }
        dis.emplace_back(d);
        for (auto [c, w] : edge[p]) {
            if (c != fa && !vis[c]) {
                self(c, p, d + w);
            }
        }
    };  

    std::vector<int> ans(q + 1);
    auto work = [&] (this auto &&self, int p) -> void {
        vis[p] = 1;
        std::vector<int> saved; 
        has[0] = 1;
        saved.emplace_back(0); 
        for (auto [c, w] : edge[p]) {
            if (!vis[c]) {
                dis.clear();
                update(c, p, w);
                for (auto x : dis) {
                    for (int i = 1; i <= q; i++) {
                        if (qry[i] >= x) {
                            ans[i] |= has[qry[i] - x];
                        }
                    }
                }
                for (auto c : dis) {
                    if (!has[c]) { 
                        has[c] = 1;
                        saved.emplace_back(c);
                    }
                }
            }
        }
        for (auto x : saved) {
            has[x] = 0;
        }

        for (auto [c, w] : edge[p]) {
            if (!vis[c]) {
                self(findCd(c));
            }
        }
    };

    work(findCd(1));

    for (int i = 1; i <= q; i++) {
        std::cout << (ans[i] ? "AYE" : "NAY") << "\n";
    }
    return ;
}

int main() {
    std::cin.tie(nullptr) -> std::ios_base::sync_with_stdio(false);

    int t = 1;
    // std::cin >> t;

    while (t-->0) {
        solve();
    }

    return 0;
}
